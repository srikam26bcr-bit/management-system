const form = document.getElementById("studentForm"); const table = document.getElementById("studentTable"); const search = document.getElementById("search"); const count = document.getElementById("studentCount"); const emptyMessage = document.getElementById("emptyMessage"); const editId = document.getElementById("editId"); const submitBtn = document.getElementById("submitBtn"); const formTitle = document.getElementById("formTitle"); const cancelBtn = document.getElementById("cancelBtn"); let students = []; try { students = JSON.parse( localStorage.getItem("students") || "[]" ); if (!Array.isArray(students)) students = []; } catch { students = []; } function saveStudents() { try { localStorage.setItem("students", JSON.stringify(students)); } catch { alert("Unable to save data in this browser."); } } function renderStudents() { table.replaceChildren(); const keyword = search.value.trim().toLowerCase(); const filtered = students.filter(student => [student.studentId, student.name, student.department, student.year, student.email] .some(value => value.toLowerCase().includes(keyword)) ); count.textContent = `Total Students: ${students.length}`; emptyMessage.hidden = filtered.length > 0; filtered.forEach(student => { const row = document.createElement("tr"); [ student.studentId, student.name, student.department, student.year, student.email ].forEach(value => { const cell = document.createElement("td"); cell.textContent = value; row.appendChild(cell); }); const actions = document.createElement("td"); const editButton = document.createElement("button"); editButton.textContent = "Edit"; editButton.className = "edit-btn"; editButton.addEventListener("click", () => { editStudent(student.studentId); }); const deleteButton = document.createElement("button"); deleteButton.textContent = "Delete"; deleteButton.className = "delete-btn"; deleteButton.addEventListener("click", () => { deleteStudent(student.studentId); }); actions.append(editButton, deleteButton); row.appendChild(actions); table.appendChild(row); }); } form.addEventListener("submit", function(event) { event.preventDefault(); const student = { studentId: document.getElementById("studentId").value.trim(), name: document.getElementById("name").value.trim(), department: document.getElementById("department").value, year: document.getElementById("year").value, email: document.getElementById("email").value.trim() }; if (Object.values(student).some(value => !value)) { alert("Please fill all fields."); return; } const duplicateId = students.some(item => item.studentId.toLowerCase() === student.studentId.toLowerCase() && item.studentId !== editId.value ); if (duplicateId) { alert("Student ID already exists!"); return; } if (editId.value) { students = students.map(item => item.studentId === editId.value ? student : item ); alert("Student updated successfully!"); } else { students.push(student); alert("Student added successfully!"); } saveStudents(); form.reset(); resetEdit(); renderStudents(); }); function editStudent(id) { const student = students.find(item => item.studentId === id); if (!student) return; document.getElementById("studentId").value = student.studentId; document.getElementById("name").value = student.name; document.getElementById("department").value = student.department; document.getElementById("year").value = student.year; document.getElementById("email").value = student.email; // Keep the original ID for identifying the record. editId.value = student.studentId; document.getElementById("studentId").readOnly = true; formTitle.textContent = "Edit Student"; submitBtn.textContent = "Update Student"; cancelBtn.hidden = false; window.scrollTo({ top: 0, behavior: "smooth" }); } function deleteStudent(id) { if (!confirm("Are you sure you want to delete this student?")) { return; } students = students.filter(item => item.studentId !== id); saveStudents(); if (editId.value === id) { form.reset(); resetEdit(); } renderStudents(); } function resetEdit() { editId.value = ""; document.getElementById("studentId").readOnly = false; formTitle.textContent = "Add Student"; submitBtn.textContent = "Add Student"; cancelBtn.hidden = true; } cancelBtn.addEventListener("click", () => { form.reset(); resetEdit(); }); search.addEventListener("input", renderStudents); renderStudents();
